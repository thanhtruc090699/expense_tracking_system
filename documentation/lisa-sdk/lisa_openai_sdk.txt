"""
Access Lisa (lisa-pro-03-2026) using the OpenAI SDK.

Requires: pip install openai
"""

import os
from openai import OpenAI


def main():
    api_key = "sk-xxxxxxxxxxx"  #replace with your own key, you can get that via your profile on the LISA UI

    client = OpenAI(base_url="https://chat-1.ki-awz.iisys.de/api", api_key=api_key)

    response = client.chat.completions.create(
        model="lisa-pro-03-2026", #or whatever other LISA model you want to use
        messages=[
            {"role": "system", "content": "You are a helpful assistant."},
            {"role": "user", "content": "Tell me a funny joke!"},
        ],
    )

    print(response.choices[0].message.content)


if __name__ == "__main__":
    main()
